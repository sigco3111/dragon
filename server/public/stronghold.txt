// Stronghold browser (§4.2, N3): chapter city troops/prod bounded edits plus
// the read-only reference/delete-blocker report. Base cities stay 192-locked
// (no add/edit/delete of base records); only chapter-state troops/prod within
// the record's own caps are writable. Saves use absolute values, so a retry
// with a fresh key is safe (re-applying identical values is a no-op server
// side). Unknown-outcome failures keep inputs on screen and say so.
export function installStronghold({ api, authority, work, isBusy }) {
  const q = (id) => document.querySelector('#' + id);
  let identity = null, games = [], selected = null, detail = null, summary = null, chapterId = null;
  const allowed = () => Boolean(identity?.user) && !identity.user.mustChangePassword;
  const message = (text) => { q('stronghold-message').textContent = text; };
  function refsFor(idx) {
    return (detail?.strongholds ?? []).find(r => r.runtimeSlot === idx) ?? null;
  }
  function blockers(r) {
    const list = [];
    if (!r) return ['查無引用資料，先讀取管理清單。'];
    if (r.placed) list.push('已放入地圖，須先從地圖移除');
    for (const road of r.roads ?? []) list.push(`道路引用：${road}`);
    for (const chapter of r.chapters ?? []) list.push(`章節引用：${chapter}`);
    list.push('基礎據點鎖定 192 座，不開放增刪');
    return list;
  }
  function chapter() {
    return (detail?.chapters ?? []).find(c => c.chapterId === chapterId) ?? null;
  }
  function render() {
    q('stronghold-editor').hidden = !allowed();
    q('stronghold-games').replaceChildren();
    for (const game of games) {
      const option = document.createElement('option');
      option.value = game.gameId; option.textContent = game.metadata.name + '／草稿 ' + game.draftRevision;
      q('stronghold-games').append(option);
    }
    if (games.some((game) => game.gameId === selected)) q('stronghold-games').value = selected;
    q('stronghold-chapters').replaceChildren();
    for (const c of detail?.chapters ?? []) {
      const option = document.createElement('option');
      option.value = c.chapterId; option.textContent = (c.name ?? c.chapterId);
      q('stronghold-chapters').append(option);
    }
    if ((detail?.chapters ?? []).some((c) => c.chapterId === chapterId)) q('stronghold-chapters').value = chapterId;
    q('stronghold-cities').replaceChildren();
    const st = chapter();
    if (!st) return;
    for (const c of st.cities ?? []) {
      const row = document.createElement('div'), label = document.createElement('p');
      const r = refsFor(c.idx);
      label.textContent = `#${c.idx} ${c.name}／勢力 ${c.faction ?? '—'}／道路 ${(r?.roads ?? []).length}／章節 ${(r?.chapters ?? []).length}`;
      row.append(label);
      for (const [field, cap] of [['troops', c.troopsCap], ['prod', c.maxProd]]) {
        const lab = document.createElement('label');
        lab.textContent = field + '（上限 ' + (cap ?? '?') + '）';
        const input = document.createElement('input');
        input.type = 'number'; input.min = '0';
        if (Number.isSafeInteger(cap)) input.max = String(cap);
        input.value = String(c[field] ?? '');
        input.dataset.cityIdx = String(c.idx); input.dataset.field = field;
        lab.append(input); row.append(lab);
      }
      const save = document.createElement('button');
      save.type = 'button'; save.textContent = '保存據點數值'; save.disabled = isBusy();
      save.onclick = () => work(() => saveCity(c.idx));
      row.append(save);
      const del = document.createElement('button');
      del.type = 'button'; del.textContent = '刪除檢查';
      del.disabled = isBusy();
      del.onclick = () => work(async () => {
        message(`「${c.name}」不可刪除：${blockers(r).join('；')}。`);
      });
      row.append(del);
      q('stronghold-cities').append(row);
    }
  }
  function readInputs(idx) {
    const out = {};
    for (const input of document.querySelectorAll('#stronghold-cities input')) {
      if (Number(input.dataset.cityIdx) !== idx) continue;
      const value = input.value.trim() === '' ? NaN : Number(input.value);
      out[input.dataset.field] = value;
    }
    return out;
  }
  async function saveCity(idx) {
    const st = chapter();
    const current = (st?.cities ?? []).find(c => c.idx === idx);
    if (!current || !summary || !detail) throw new Error('請先選擇遊戲與章節。');
    const values = readInputs(idx), patch = {};
    for (const field of ['troops', 'prod']) {
      const value = values[field], cap = field === 'troops' ? current.troopsCap : current.maxProd;
      if (typeof value !== 'number' || !Number.isInteger(value)) throw new Error(`${field}須為整數。`);
      if (value < 0) throw new Error(`${field}不可為負。`);
      if (!Number.isSafeInteger(cap)) throw new Error('上限未知，拒絕寫入。');
      if (value > cap) throw new Error(`${field}超過上限 ${cap}，已攔截未發送。`);
      if (value !== current[field]) patch[field] = value;
    }
    if (!Object.keys(patch).length) { message('與目前值相同，未發送。'); return; }
    let result;
    try {
      result = await api('/api/games/' + encodeURIComponent(summary.gameId) + '/draft/save',
        { metadata: detail.metadata, resourceEdits: [], chapterEdits: [{ op: 'city', chapterId, cityIdx: idx, values: patch }] },
        { key: crypto.randomUUID(), revision: summary.draftRevision });
    } catch (error) {
      message(error instanceof Error ? error.message : String(error));
      return;
    }
    const stored = games.find((game) => game.gameId === summary.gameId);
    if (stored) stored.draftRevision = result.draftRevision;
    await inspect();
    message(`已保存城 #${idx}，草稿修訂 ${result.draftRevision}。`);
  }
  async function refresh() {
    if (!allowed()) throw new Error('請先登入。');
    const result = await api('/api/games');
    if (!Array.isArray(result.games)) throw new Error('遊戲清單回應無效。');
    games = result.games; selected = games[0]?.gameId ?? null; detail = null; summary = null; chapterId = null; render();
    if (selected) await inspect();
  }
  async function inspect() {
    const gameId = q('stronghold-games').value;
    summary = games.find((game) => game.gameId === gameId);
    if (!summary) throw new Error('請先選擇遊戲。');
    selected = gameId;
    detail = await api('/api/games/' + encodeURIComponent(gameId) + '/draft?revision=' + encodeURIComponent(summary.draftRevision));
    const chapters = detail?.chapters ?? [];
    if (!chapters.some((c) => c.chapterId === chapterId)) chapterId = chapters[0]?.chapterId ?? null;
    render();
    message('據點數值僅 troops／prod 可寫且受上限門；基礎 192 鎖定不開放增刪。');
  }
  q('stronghold-refresh').onclick = () => work(refresh);
  q('stronghold-inspect').onclick = () => work(inspect);
  q('stronghold-chapters').onchange = () => { chapterId = q('stronghold-chapters').value; render(); };
  return Object.freeze({ render, setIdentity(value) { identity = value; if (!allowed()) { games = []; selected = null; detail = null; summary = null; chapterId = null; } render(); }, canLeave() { return true; } });
}

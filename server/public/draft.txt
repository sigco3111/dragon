// UI state is not authority. Every read/write uses the authenticated fixed-copy API.
// Vue receives display projections only; credentials and operation bodies stay here.
export function installDraftEditor({ api, authority, work, isBusy, publish, onContextChanged = () => {}, storage = globalThis.sessionStorage, confirm = globalThis.confirm, prompt = globalThis.prompt, openTrial = () => window.open('/api/trial/wait', '_blank') }) {
  const uuid = value => typeof value === 'string' && /^[a-f0-9]{8}-[a-f0-9]{4}-4[a-f0-9]{3}-[89ab][a-f0-9]{3}-[a-f0-9]{12}$/.test(value), revision = value => typeof value === 'string' && /^[1-9][0-9]{0,63}$/.test(value);
  let owner = null, identity = null, generation = 0, games = [], loaded = null, attempt = null, chapters = [], resources = [], message = '請查看自己的遊戲並明確載入已保存修訂。';
  const fields = { gameId: '', revision: '', name: '', introduction: '', chapterId: '' };
  const input = () => ({ name: fields.name, introduction: fields.introduction });
  const dirty = () => Boolean(loaded && (fields.name !== loaded.metadata.name || fields.introduction !== loaded.metadata.introduction));
  const unresolved = () => Boolean(attempt && !['failed', 'revoked'].includes(attempt.state));
  const permitted = () => Boolean(identity && identity === authority() && !identity.user.mustChangePassword);
  const storageKey = () => 'editor-draft-operation:' + owner;
  function persist() { try { if (attempt) storage.setItem(storageKey(), JSON.stringify({ gameId: attempt.gameId, revision: attempt.revision, key: attempt.key })); else storage.removeItem(storageKey()); } catch { message = '無法保留操作鍵；請勿在保存結果確認前重新載入。'; } }
  function restore() {
    try { const text = storage.getItem(storageKey()); if (!text) return; if (text.length > 512) throw new Error('invalid operation association'); const data = JSON.parse(text); if (!uuid(data.gameId) || !revision(data.revision) || typeof data.key !== 'string' || !/^[A-Za-z0-9_-]{16,128}$/.test(data.key) || Object.keys(data).sort().join(',') !== 'gameId,key,revision') throw new Error('invalid operation association'); attempt = { ...data, state: 'unknown', body: null }; message = '上次保存結果未確認；先查詢原操作。此頁沒有原請求內容，不能重造同鍵請求。'; }
    catch { message = '操作鍵關聯無法讀取；不視為已保存，也不自動提交。'; }
  }
  function render() {
    const allowed = permitted(), locked = !allowed || isBusy();
    publish({
      allowed, fields: allowed ? { ...fields } : { gameId: '', revision: '', name: '', introduction: '', chapterId: '' },
      games: allowed ? games.map(game => ({ gameId: game.gameId, name: game.metadata.name, draftRevision: game.draftRevision })) : [],
      chapters: allowed ? chapters.map(chapter => ({ ...chapter, gaps: { ...chapter.gaps } })) : [],
      resources: allowed ? resources.map(chapter => ({ ...chapter, factions: chapter.factions.map(faction => ({ ...faction })) })) : [],
      message: allowed ? message : '',
      context: allowed && loaded ? { gameId: loaded.gameId, name: loaded.metadata.name, draftRevision: loaded.draftRevision, dirty: dirty(), verified: Boolean(loaded.verifiedRead), pending: Boolean(attempt) } : null,
      locked, gameLocked: locked || Boolean(attempt), loadLocked: locked || unresolved(), editLocked: locked || !loaded?.verifiedRead || Boolean(attempt),
      trialLocked: locked || !loaded?.verifiedRead || Boolean(attempt) || chapters.length === 0,
      queryLocked: locked || !attempt, retryLocked: locked || !attempt?.body || !['failed', 'not-found'].includes(attempt.state),
    });
    onContextChanged();
  }
  function canLeave(forUnload = false) { return !(dirty() || attempt) || (!forUnload && confirm('有未保存資料或未確認的保存操作。確定離開？未確認操作仍須查詢原鍵，不會自動重試。')); }
  function clearFields() { Object.assign(fields, { gameId: '', revision: '', name: '', introduction: '', chapterId: '' }); }
  function setIdentity(next) {
    if (identity === next) { render(); return; } generation++; identity = next;
    if (next && owner !== next.user.id) { owner = next.user.id; games = []; loaded = null; attempt = null; chapters = []; resources = []; clearFields(); restore(); }
    else if (next && !attempt) restore();
    // Expiry hides the projection, not falsely acknowledging or resubmitting input.
    render();
  }
  function ticket() { const user = authority(), serial = ++generation; return () => user === authority() && user === identity && serial === generation; }
  function metadata(value) { if (!value || typeof value.name !== 'string' || typeof value.introduction !== 'string') throw new Error('資料回應無效。'); return { name: value.name, introduction: value.introduction }; }
  function summary(result, expected) { if (!result || result.gameId !== expected.gameId || !revision(result.draftRevision) || ![expected.revision, String(BigInt(expected.revision) + 1n)].includes(result.draftRevision)) throw new Error('保存回應身份或修訂無效，請查詢原操作。'); return metadata(result.metadata); }
  function accept(result) { const value = summary(result, attempt); loaded = { ...loaded, gameId: result.gameId, draftRevision: result.draftRevision, metadata: value }; fields.name = value.name; fields.introduction = value.introduction; fields.revision = result.draftRevision; attempt = null; persist(); message = '已確認保存修訂 ' + result.draftRevision + '；沒有切換最新版本。'; }
  async function refreshGames() {
    const valid = ticket(), response = await api('/api/games'); if (!valid()) return;
    if (!Array.isArray(response.games) || response.games.some(game => !uuid(game.gameId) || !revision(game.draftRevision))) throw new Error('遊戲清單回應無效。');
    const nextGames = response.games.map(game => ({ gameId: game.gameId, draftRevision: game.draftRevision, metadata: metadata(game.metadata) }));
    const selected = attempt?.gameId ?? loaded?.gameId ?? fields.gameId; games = nextGames;
    fields.gameId = games.some(game => game.gameId === selected) ? selected : games[0]?.gameId ?? '';
    if (!loaded && !attempt) fields.revision = games.find(game => game.gameId === fields.gameId)?.draftRevision ?? '';
    message = attempt ? '請先查詢未確認的原保存操作。' : '清單修訂只供選擇，載入前不替換已保存上下文。';
  }
  async function load() {
    if (unresolved()) throw new Error('未確認原保存操作，不能換目標或另起保存。');
    if (!canLeave()) return; const gameId = fields.gameId, requested = fields.revision;
    if (!uuid(gameId) || !revision(requested)) throw new Error('請選遊戲並輸入明確已保存修訂。');
    const valid = ticket(), response = await api('/api/games/' + gameId + '/draft?revision=' + encodeURIComponent(requested)); if (!valid()) return;
    if (response.gameId !== gameId || response.draftRevision !== requested || response.profileRevision !== 'fixed-copy-resources-1' || !Array.isArray(response.chapters)) throw new Error('草稿回應上下文無效。');
    const value = metadata(response.metadata);
    const nextChapters = response.chapters.map(chapter => {
      if (typeof chapter.chapterId !== 'string' || !chapter.chapterId || !chapter.gaps || !Number.isSafeInteger(chapter.gaps.saveGaps) || !Number.isSafeInteger(chapter.gaps.activeGaps) || !Array.isArray(chapter.factions)) throw new Error('章資料回應無效。');
      return { chapterId: chapter.chapterId, name: String(chapter.name || chapter.chapterId), gaps: { saveGaps: chapter.gaps.saveGaps, activeGaps: chapter.gaps.activeGaps } };
    });
    const nextResources = response.chapters.map((chapter, index) => ({ chapterId: nextChapters[index].chapterId, name: nextChapters[index].name, factions: chapter.factions.map(faction => ({ slot: faction.slot, money: faction.money, reserve_cav: faction.reserve_cav, reserve_arc: faction.reserve_arc, reserve_inf: faction.reserve_inf })) }));
    loaded = { gameId, draftRevision: requested, metadata: value, verifiedRead: true }; attempt = null; persist(); chapters = nextChapters; resources = nextResources;
    fields.name = value.name; fields.introduction = value.introduction;
    if (!chapters.some(chapter => chapter.chapterId === fields.chapterId)) fields.chapterId = chapters[0]?.chapterId ?? '';
    message = '已載入明確修訂；只有名稱／簡介可編輯，四資源只讀。';
  }
  function normalized() { const value = input(); for (const [name, limit] of [['name', 8], ['introduction', 20]]) { if (!value[name].isWellFormed() || [...value[name]].some(c => c.codePointAt(0) < 32 || c.codePointAt(0) >= 127 && c.codePointAt(0) <= 159)) throw new Error('名稱／簡介含無效字元。'); value[name] = value[name].trim().normalize('NFC'); if ([...value[name]].length > limit || name === 'name' && !value[name].length) throw new Error('名稱最多8字元，簡介最多20字元。'); } return value; }
  async function submit(retry = false) {
    if (!loaded?.verifiedRead || !permitted()) throw new Error('請先載入自己的精確草稿。');
    if (!retry) { if (attempt) throw new Error('先查詢原保存，不另起鍵。'); attempt = { gameId: loaded.gameId, revision: loaded.draftRevision, key: crypto.randomUUID(), body: { metadata: normalized(), resourceEdits: [] }, state: 'unknown' }; persist(); }
    if (!attempt?.body || retry && !['failed', 'not-found'].includes(attempt.state)) throw new Error('沒有原請求內容或尚未確認可重試；不重造同鍵內容。');
    await sendAttempt('正在保存；如失去回應，保留原輸入與操作鍵並查詢。');
  }
  async function sendAttempt(progress) {
    const pending = attempt, valid = ticket(); pending.state = 'unknown'; message = progress; render();
    try { const result = await api('/api/games/' + pending.gameId + '/draft/save', pending.body, { key: pending.key, revision: pending.revision }); if (valid() && attempt === pending) accept(result); }
    catch (error) { if (valid() && attempt === pending) message = '保存未確認：' + error.message + ' 原輸入保留，請查詢原操作；不自動重試。'; throw error; }
  }
  async function query() {
    if (!attempt) return; const pending = attempt, valid = ticket();
    try { const response = await api('/api/games/' + pending.gameId + '/draft/operations/' + encodeURIComponent(pending.key)); if (!valid() || attempt !== pending) return;
      if (response.state === 'committed') accept(response.result);
      else if (['pending', 'failed'].includes(response.state) && response.gameId === pending.gameId && response.expectedRevision === pending.revision) { pending.state = response.state; message = response.state === 'pending' ? '服務仍pending；保留原鍵，不另起保存。' : '服務已記錄失敗：' + response.error + '。輸入保留，可明確重試原請求或確認放棄後重新載入。'; }
      else throw new Error('操作回應上下文無效。');
    } catch (error) { if (valid() && attempt === pending) { if (error.status === 404) { if (pending.state !== 'revoked') pending.state = 'not-found'; message = pending.state === 'revoked' ? '舊會話已由實際改密撤銷；查無原操作仍不宣稱已保存或失敗。可確認放棄後載入明確修訂。' : '目前查無原操作，不保證在途請求已結束；有原內容才可明確重試相同原鍵。不保留原內容時，可明確改密撤銷所有舊會話後再載入，不能重造請求或自動另起鍵。'; } else if (error.code === 'OPERATION_REVOKED') { pending.state = 'revoked'; message = '原操作會話已撤銷，不能復活舊鍵；確認放棄後載入明確修訂。'; } else { message = '查詢未確認：' + error.message; throw error; } } }
  }
  function action(callback) { return work(async () => { if (!permitted()) throw new Error('請先登入並完成改密。'); try { return await callback(); } finally { render(); } }); }
  async function startTrial(child) {
    if (!loaded?.verifiedRead || attempt) throw new Error('請先載入自己的精確草稿並確認無未確認操作。');
    const chapterId = fields.chapterId;
    if (!chapters.some(chapter => chapter.chapterId === chapterId)) throw new Error('請選擇要試運行的章節。');
    const gameId = loaded.gameId, expected = loaded.draftRevision, valid = ticket();
    const result = await api('/api/games/' + gameId + '/trials', { expectedRevision: expected, chapterId });
    if (!valid()) { try { if (!child.closed) child.close(); } catch { /* retain */ } return; }
    if (child.closed) throw new Error('試運行視窗已關閉，請重新開啟。');
    if (result?.state !== 'active' || !uuid(result.trialId) || result.gameId !== gameId || result.draftRevision !== expected || result.chapterId !== chapterId || result.ownerId !== identity.user.id || typeof result.manifestDigest !== 'string' || !/^[a-f0-9]{64}$/.test(result.manifestDigest)) throw new Error('試運行回應身份不符。');
    child.location.replace('/api/trial/web/?trial=' + result.trialId);
    message = '已開啟草稿試運行視窗；僅記憶體、進度不保存。';
  }
  function trial() {
    if (isBusy() || !permitted() || !loaded?.verifiedRead || attempt || !chapters.length) return;
    const child = openTrial();
    if (!child) { message = '瀏覽器攔截了彈出視窗；請允許後再點擊「測試運行本章」重試。'; render(); return; }
    try { child.opener = null; } catch { /* cross-origin timing */ }
    return action(async () => { try { await startTrial(child); } catch (error) { try { if (!child.closed) child.close(); } catch { /* retain original failure */ } throw error; } });
  }
  function chooseGame(gameId) { if (isBusy()) { render(); return; } selectGame(gameId); }
  function selectGame(gameId) {
    if (!permitted() || attempt || !canLeave()) { render(); return; }
    if (!games.some(game => game.gameId === gameId)) return;
    fields.gameId = gameId; loaded = null; chapters = []; resources = []; fields.name = ''; fields.introduction = ''; fields.chapterId = ''; fields.revision = games.find(game => game.gameId === gameId).draftRevision; generation++; render();
  }
  function setField(name, value) {
    if (!permitted() || isBusy() || typeof value !== 'string') return;
    if (name === 'revision' && !unresolved() || ['name', 'introduction'].includes(name) && loaded?.verifiedRead && !attempt || name === 'chapterId' && chapters.some(chapter => chapter.chapterId === value)) { fields[name] = value; render(); }
  }
  async function gotoGame(gameId) { if (!permitted()) return; if (!games.some(game => game.gameId === gameId)) await refreshGames(); if (!games.some(game => game.gameId === gameId)) throw new Error('草稿清單沒有此遊戲。'); selectGame(gameId); }
  // §4.5 chapter ops: each op is its own save (same attempt discipline).
  async function submitChapters(edits) {
    if (!loaded?.verifiedRead || !permitted()) throw new Error('請先載入自己的精確草稿。');
    if (attempt) throw new Error('先查詢原保存，不另起鍵。');
    attempt = { gameId: loaded.gameId, revision: loaded.draftRevision, key: crypto.randomUUID(), body: { metadata: normalized(), resourceEdits: [], chapterEdits: edits }, state: 'unknown' }; persist();
    await sendAttempt('正在保存章節修改；如失去回應，保留原輸入與操作鍵並查詢。');
  }
  async function chapterAction(edit) { await submitChapters([edit]); await load(); await refreshGames(); }
  function chapter(op, chapterId) {
    return action(async () => {
      if (op === 'add-blank') return chapterAction({ op });
      const selected = chapters.find(item => item.chapterId === chapterId);
      if (!selected || !['copy', 'rename', 'delete'].includes(op)) throw new Error('請選擇已載入的章節。');
      const edit = { op, chapterId };
      if (op === 'rename') { edit.name = prompt('章節新名稱', selected.name); if (edit.name === null) return; }
      if (op === 'delete' && !confirm('刪除章節「' + selected.name + '」？僅影響草稿，不改變已發布版本；草稿可暫無章節，但屆時禁止發布。')) return;
      return chapterAction(edit);
    });
  }
  render();
  return {
    actions: { refresh: () => action(refreshGames), load: () => action(load), save: () => action(() => submit()), query: () => action(query), retry: () => action(() => submit(true)), chooseGame, setField, chapter, trial },
    setIdentity, render, canLeave, gotoGame,
    savedContext() { return permitted() && loaded?.verifiedRead && !attempt ? { gameId: loaded.gameId, draftRevision: loaded.draftRevision, dirty: dirty() } : null; },
    passwordChanged(next) { if (next !== authority() || next?.user.id !== owner || !attempt) return; attempt.state = 'revoked'; message = '服務已確認改密並撤銷舊會話。這不代表原保存成功或失敗；確認放棄後載入明確修訂。'; render(); },
    clearForLogout() { generation++; loaded = null; attempt = null; chapters = []; resources = []; games = []; clearFields(); render(); },
  };
}

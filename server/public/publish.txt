// Author release UI only. Publish pins an author-confirmed version; listing stays admin-only.
export function installPublish({api,authority,work,isBusy}) {
 const q=id=>document.querySelector('#'+id), uuid=v=>typeof v==='string'&&/^[a-f0-9]{8}-[a-f0-9]{4}-4[a-f0-9]{3}-[89ab][a-f0-9]{3}-[a-f0-9]{12}$/.test(v), revision=v=>typeof v==='string'&&v.length<=8192&&/^[1-9][0-9]*$/.test(v);
 const errors={PUBLISH_NAME:'發布名稱無效（須非空且最多40字元）。',PUBLISH_REVISION:'草稿修訂無效。',PUBLISH_STALE:'草稿已變更，請重新載入再發布。',PUBLISH_QUERY:'發布請求格式無效。',RELEASE_QUERY:'版本查詢格式無效。',GAME_NOT_FOUND:'遊戲不存在或非本人所有。',BUILTIN_PROTECTED:'內置遊戲不可發布。',LOGIN_REQUIRED:'請登入。',ANNOUNCE_CURSOR:'公告遊標無效。'};
 let identity=null,generation=0,games=[],versions=[],announcement=null,announced=false,published=null;
 const allowed=()=>Boolean(identity?.user)&&!identity.user.mustChangePassword&&(identity.user.role==='author'||identity.user.role==='admin');
 const message=text=>{q('publish-message').textContent=text;};
 function gameList(value){if(!value||!Array.isArray(value.games)||value.games.length>10000)throw new Error('游戏清單格式無效。');const seen=new Set();return value.games.map(g=>{if(!g||!uuid(g.gameId)||seen.has(g.gameId)||!g.metadata||(g.metadata.name!==null&&typeof g.metadata.name!=='string')||(g.metadata.introduction!==null&&typeof g.metadata.introduction!=='string')||typeof g.listed!=='boolean'||typeof g.draftRevision!=='string'||!revision(g.draftRevision))throw new Error('游戏清單欄位無效。');seen.add(g.gameId);return Object.freeze({gameId:g.gameId,name:g.metadata.name,introduction:g.metadata.introduction,listed:g.listed,draftRevision:g.draftRevision});});}
 function versionList(value){if(!value||!Array.isArray(value.releases)||value.releases.length>100)throw new Error('版本清單格式無效。');return value.releases.map(r=>{if(!r||typeof r.releaseId!=='string'||!uuid(r.releaseId)||typeof r.ordinal!=='string'||!/^[0-9]+(\.[0-9]+)+$/.test(r.ordinal)||typeof r.draftRevision!=='string'||!revision(r.draftRevision)||typeof r.createdAt!=='string')throw new Error('版本欄位無效。');return Object.freeze({releaseId:r.releaseId,ordinal:r.ordinal,draftRevision:r.draftRevision,createdAt:r.createdAt});});}
 function pin(value){if(!value||typeof value!=='object'||Array.isArray(value)||typeof value.ordinal!=='string'||!/^[0-9]+(\.[0-9]+)+$/.test(value.ordinal)||typeof value.revision!=='string'||!revision(value.revision)||typeof value.publishedAt!=='string'||(value.name!==null&&typeof value.name!=='string'))throw new Error('发布回執格式無效。');return Object.freeze({ordinal:value.ordinal,revision:value.revision,publishedAt:value.publishedAt,name:value.name});}
 function announcementList(value){if(!value||!Array.isArray(value.announcements))throw new Error('公告清單格式無效。');const hit=value.announcements.find(a=>a&&a.gameId===selected());if(!hit)return null;if(typeof hit.ordinal!=='string'||typeof hit.name!=='string'||typeof hit.creatorAccount!=='string'||typeof hit.firstOnlineAt!=='string')throw new Error('公告欄位無效。');return Object.freeze({ordinal:hit.ordinal,name:hit.name,creatorAccount:hit.creatorAccount,firstOnlineAt:hit.firstOnlineAt});}
 function selected(){const id=q('publish-game').value;return uuid(id)?id:null;}
 function render(){q('publish-editor').hidden=!allowed();if(!allowed())return;const gameSelect=q('publish-game'),keep=selected();gameSelect.replaceChildren();for(const g of games){const option=document.createElement('option');option.value=g.gameId;option.textContent=(g.name??'未命名')+'／草稿'+g.draftRevision+'／'+(g.listed?'已上架':'未上架');gameSelect.append(option);}if(keep&&games.some(g=>g.gameId===keep))gameSelect.value=keep;const current=games.find(g=>g.gameId===selected());q('publish-name').value=current?.name??'';const box=q('publish-versions');box.replaceChildren();if(versions.length===0){const empty=document.createElement('p');empty.textContent='尚無正式版本。';box.append(empty);}for(const v of versions){const row=document.createElement('p');row.textContent='版本 '+v.ordinal+'／草稿修訂 '+v.draftRevision+'／定版 '+v.createdAt;box.append(row);}const result=q('publish-result');result.replaceChildren();if(published){const done=document.createElement('p');done.textContent='已定版：版本 '+published.ordinal+'／修訂 '+published.revision+'／'+published.publishedAt+'（發布≠上架，玩家目錄需管理員上架；舊版本已替換刪除。）';result.append(done);}const ann=q('publish-announcement');ann.replaceChildren();if(announced){const view=document.createElement('p');view.textContent=announcement?('公告：'+announcement.name+'／版本 '+announcement.ordinal+'／作者 '+announcement.creatorAccount+'／首次上架 '+announcement.firstOnlineAt):'無公告：遊戲未上架，發布只定版不公開。';ann.append(view);}const disabled=!allowed()||isBusy()||!selected();q('publish-games-refresh').disabled=!allowed()||isBusy();q('publish-versions-refresh').disabled=disabled;q('publish-button').disabled=disabled;q('publish-ann-preview').disabled=disabled;}
 function current(ctx){const live=authority();return allowed()&&live?.user.id===ctx.id&&live.user.version===ctx.version&&!live.user.mustChangePassword&&generation===ctx.generation;}
 function capture(){if(!allowed())throw new Error('請先登入作者帳戶。');return Object.freeze({id:identity.user.id,version:identity.user.version,generation});}
 async function refresh(){const ctx=capture();const result=await api('/api/games');if(!current(ctx))return;games=gameList(result);versions=[];announcement=null;announced=false;published=null;render();}
 async function versionsRefresh(){const ctx=capture(),id=selected();if(!id)throw new Error('請先選擇遊戲。');const result=await api('/api/games/'+encodeURIComponent(id)+'/releases');if(!current(ctx)||selected()!==id)return;versions=versionList(result);render();}
 async function publish(event){event.preventDefault();const ctx=capture(),id=selected();if(!id)throw new Error('請先選擇遊戲。');const name=q('publish-name').value;if(!confirm('定版發布此遊戲？新版本將替換刪除舊版本；發布≠上架，玩家目錄仍需管理員上架。\n'+id))return;try{const value=await api('/api/publish',{gameId:id,name});if(!current(ctx)||selected()!==id)return;published=pin(value);message('');await versionsRefresh();}catch(error){if(current(ctx))message(errors[error.code]??error.message);throw error;}}
 async function preview(){const ctx=capture(),id=selected();if(!id)throw new Error('請先選擇遊戲。');const result=await api('/api/announcements');if(!current(ctx)||selected()!==id)return;announcement=announcementList(result);announced=true;render();}
 q('publish-games-refresh').onclick=()=>work(refresh);q('publish-versions-refresh').onclick=()=>work(versionsRefresh);q('publish-form').onsubmit=event=>work(()=>publish(event));q('publish-ann-preview').onclick=()=>work(preview);
 return Object.freeze({render,setIdentity(value){const changed=identity?.user.id!==value?.user.id||identity?.user.version!==value?.user.version||identity?.user.role!==value?.user.role||identity?.user.mustChangePassword!==value?.user.mustChangePassword;identity=value;if(changed){generation++;games=[];versions=[];announcement=null;announced=false;published=null;}render();},canLeave(){return true;}});
}
